### Task 1: traffic.py — the floor's ledger (pure)

**Files:** Create `m6/fleet/traffic.py`; Test `m6/tests/test_traffic.py`.

**Interfaces (normative):**

```python
# Elements: a node is the graph's (x, y) tuple; an edge is a frozenset
# {(x1,y1), (x2,y2)} — undirected ON PURPOSE (one corridor segment is
# one piece of floor whichever way you drive it; that identity is what
# catches the head-on case).

def edge(a, b) -> frozenset            # the undirected element
def route_elements(points) -> list     # [node0, edge01, node1, edge12, ...]
                                       # interleaved, in travel order

class Reservations:
    def hold(self, vehicle, elements) -> list
        # Grant the longest CONTIGUOUS PREFIX of `elements` whose members
        # are free or already this vehicle's. Never grants a hole. Returns
        # the granted prefix (possibly []). Records waiting_on[vehicle] =
        # the first ungranted element (or clears it when all granted).
    def release_through(self, vehicle, node) -> None
        # Free everything this vehicle holds up to AND INCLUDING the edge
        # that arrives at `node`, keeping `node` itself and everything
        # after. A node the vehicle never held is a no-op.
    def release_all(self, vehicle, keep=None) -> None
        # Free everything; `keep` (a node) stays held — the truck is
        # standing on it.
    def owner_of(self, element) -> str | None
    def held_by(self, vehicle) -> list
    def waiting_on(self, vehicle) -> element | None
    def find_cycle(self) -> list | None
        # A cycle in the wait-for graph (vehicle -> owner of what it
        # wants), as a list of serials, or None.
    def resolve_deadlock(self, ages) -> str | None
        # ages: {vehicle: task_submit_time}. If find_cycle(), the
        # YOUNGEST vehicle in it (largest age value) releases all but the
        # node it stands on (tracked as `standing[vehicle]`), is marked
        # yielded, and its serial is returned. Else None.
    def set_standing(self, vehicle, node) -> None
    def yielded(self, vehicle) -> bool
    def clear_yield(self, vehicle) -> None
```

- [ ] **Step 1: Write the failing tests** — `m6/tests/test_traffic.py`:

```python
"""traffic.py - the floor's ledger. Pure: no MQTT, no ROS, no clock.

Reservation is PROCESS deconfliction. It is not an anti-collision
system and these tests assert nothing about safety: the scanners, the
F-model and the onboard guards are what stop a truck.
"""
import os
import sys

_HERE = os.path.dirname(os.path.abspath(__file__))
sys.path.insert(0, os.path.normpath(os.path.join(_HERE, "..", "fleet")))

import traffic as tr

A, B, C, D = (0.0, 0.0), (1.0, 0.0), (2.0, 0.0), (3.0, 0.0)


def test_an_edge_is_the_same_floor_in_both_directions():
    assert tr.edge(A, B) == tr.edge(B, A)


def test_route_elements_interleave_nodes_and_edges_in_travel_order():
    assert tr.route_elements([A, B, C]) == [
        A, tr.edge(A, B), B, tr.edge(B, C), C]


def test_a_free_route_is_granted_whole():
    r = tr.Reservations()
    want = tr.route_elements([A, B, C])
    assert r.hold("f1", want) == want
    assert r.waiting_on("f1") is None


def test_a_taken_element_truncates_the_grant_and_records_the_wait():
    r = tr.Reservations()
    r.hold("f2", [C])
    want = tr.route_elements([A, B, C])
    got = r.hold("f1", want)
    assert got == [A, tr.edge(A, B), B]          # stops before C
    assert r.waiting_on("f1") == C
    assert r.owner_of(C) == "f2"


def test_a_grant_never_has_a_hole():
    r = tr.Reservations()
    r.hold("f2", [B])
    got = r.hold("f1", tr.route_elements([A, B, C]))
    assert got == [A]                             # not [A, ..., C]


def test_head_on_is_caught_by_the_undirected_edge():
    r = tr.Reservations()
    r.hold("f1", tr.route_elements([A, B, C]))
    got = r.hold("f2", tr.route_elements([C, B, A]))
    assert got == []                              # C is f1's already
    assert r.waiting_on("f2") == C


def test_release_through_frees_the_past_and_keeps_the_present():
    r = tr.Reservations()
    r.hold("f1", tr.route_elements([A, B, C]))
    r.release_through("f1", B)
    assert r.owner_of(A) is None
    assert r.owner_of(tr.edge(A, B)) is None
    assert r.owner_of(B) == "f1"
    assert r.owner_of(tr.edge(B, C)) == "f1"


def test_release_all_can_keep_the_node_under_the_truck():
    r = tr.Reservations()
    r.hold("f1", tr.route_elements([A, B, C]))
    r.release_all("f1", keep=B)
    assert r.owner_of(B) == "f1" and r.owner_of(C) is None


def test_no_cycle_when_only_one_waits():
    r = tr.Reservations()
    r.hold("f1", [A, B])
    r.hold("f2", [B])                             # f2 waits on f1
    assert r.find_cycle() is None


def test_a_mutual_wait_is_a_cycle_and_the_youngest_yields():
    # f1 stands at A reaching for C; f2 stands at D reaching back for B.
    # Each holds what the other's next step needs. Yielding frees the
    # REACHED-FOR elements, not the ground under the yielder - which is
    # why the oldest can then move and the cycle is genuinely broken.
    r = tr.Reservations()
    r.set_standing("f1", A)
    r.set_standing("f2", D)
    r.hold("f1", tr.route_elements([A, B]))       # A, edge(A,B), B
    r.hold("f2", tr.route_elements([D, C]))       # D, edge(C,D), C
    assert r.hold("f1", tr.route_elements([A, B, C])) == \
        tr.route_elements([A, B])                 # C is f2's
    assert r.hold("f2", tr.route_elements([D, C, B])) == \
        tr.route_elements([D, C])                 # B is f1's
    assert set(r.find_cycle()) == {"f1", "f2"}
    loser = r.resolve_deadlock({"f1": 100.0, "f2": 200.0})
    assert loser == "f2"                          # younger task yields
    assert r.yielded("f2") is True
    assert r.owner_of(D) == "f2"                  # still standing there
    assert r.owner_of(C) is None                  # the reach is released
    assert r.find_cycle() is None
    assert r.hold("f1", tr.route_elements([A, B, C])) == \
        tr.route_elements([A, B, C])              # the oldest may move


def test_a_yielded_vehicle_holds_again_once_the_corridor_drains():
    r = tr.Reservations()
    r.set_standing("f1", A)
    r.set_standing("f2", D)
    r.hold("f1", tr.route_elements([A, B]))
    r.hold("f2", tr.route_elements([D, C]))
    r.hold("f1", tr.route_elements([A, B, C]))
    r.hold("f2", tr.route_elements([D, C, B]))
    assert r.resolve_deadlock({"f1": 100.0, "f2": 200.0}) == "f2"
    r.release_all("f1")                # f1 finished and left the aisle
    r.clear_yield("f2")
    assert r.yielded("f2") is False
    assert r.hold("f2", tr.route_elements([D, C, B, A])) == \
        tr.route_elements([D, C, B, A])           # the way is open now
```

### Task 2: base extension — vda_orders stitching

**Files:** Modify `m6/ipc/vda_orders.py`; Test: extend `m6/tests/test_vda_orders.py`.

**Interface change:** `accept_order(msg, current_order_id, current_update_id, executing, operating_mode)` gains an `"extend"` verdict. Rules (spec §"The M6.2 constraint this lifts"):

- Same `orderId` as executing AND `orderUpdateId == current + 1` AND the released-node prefix of the new order matches the current order's released nodes by `(nodeId, sequenceId)` for every node the current order already had **that is still released**, and the new order only ADDS released nodes (a node that was horizon may become released; a released node may not become horizon, disappear, or change position) → `("extend", "")`.
- Same orderId, wrong increment (not +1) → reject `"orderUpdateId must be exactly one more than the executing order"`.
- Same orderId, correct increment, changed prefix → reject `"an update may not change the part already driven"`.
- Different orderId while executing → reject as today.
- `orderUpdateId != 0` with NO executing order → reject as today (nothing to extend).
- Validation (`validate_order`) is unchanged and runs first.

Because `accept_order` needs the current order's nodes to compare, add a parameter: `accept_order(msg, current_order, executing, operating_mode)` where `current_order` is the full executing order dict or None — **this is a signature change; update the two call sites** (`vda_agent._on_order` and any test) and say so in the report.
